   138→// MAIN DASHBOARD
   139→// ============================================================
   140→export function PmDashboard() {
   141→  const navigate = useAppStore((s) => s.navigate);
   142→  const openApplication = useAppStore((s) => s.openApplication);
   143→  const apps = useAllApplications();
   144→  const users = useAllUsers();
   145→
   146→  // ---- KPI counts (derived, not hardcoded) ----
   147→  const kpis = React.useMemo(() => {
   148→    const total = apps.length;
   149→    const inProgress = apps.filter((a) => !["APPROVED", "REJECTED", "DRAFT"].includes(a.status)).length;
   150→    const approved = apps.filter((a) => a.status === "APPROVED").length;
   151→    const delayedAtRisk = apps.filter((a) => {
   152→      const sla = computeSLA(a);
   153→      return ["DELAYED", "CRITICAL", "AT_RISK", "BLOCKED"].includes(sla.status);
   154→    }).length;
   155→    return { total, inProgress, approved, delayedAtRisk };
   156→  }, [apps]);
   157→
   158→  // ---- SLA summary counts ----
   159→  const slaSummary = React.useMemo(() => {
   160→    const counts = { ON_TRACK: 0, AT_RISK: 0, DELAYED: 0, CRITICAL: 0, BLOCKED: 0, COMPLETED: 0 };
   161→    apps.forEach((a) => {
   162→      const sla = computeSLA(a);
   163→      if (sla.status in counts) counts[sla.status as keyof typeof counts]++;
   164→    });
   165→    return counts;
   166→  }, [apps]);
   167→
   168→  // ---- Bottleneck ----
   169→  const bottleneck = React.useMemo(() => identifyBottleneck(apps), [apps]);
   170→
   171→  return (
   172→    <div className="space-y-5">
   173→      <PageHeader
   174→        title="Project Manager Dashboard"
   175→        description="Central operational view of the building permit approval workflow."
   176→        icon={BarChart3}
   177→        breadcrumbs={[{ label: "Project Manager" }, { label: "Dashboard" }]}
   178→      />
   179→
   180→      {/* ===== KPI CARDS (4-col, compact 90-110px height) ===== */}
   181→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   182→        <CompactKpiCard icon={FileStack} value={kpis.total} label="Total Applications" accent="bg-primary/10 text-primary" onClick={() => navigate("pm-applications")} />
   183→        <CompactKpiCard icon={Clock} value={kpis.inProgress} label="In Progress" accent="bg-info/10 text-info" onClick={() => navigate("pm-applications")} />
   184→        <CompactKpiCard icon={CheckCircle2} value={kpis.approved} label="Approved" accent="bg-success/10 text-success" onClick={() => navigate("pm-applications")} />
   185→        <CompactKpiCard icon={AlertTriangle} value={kpis.delayedAtRisk} label="Delayed / At Risk" accent="bg-destructive/10 text-destructive" onClick={() => navigate("pm-sla")} />
   186→      </div>
   187→
   188→      {/* ===== Application Progress Overview (full width, searchable/filterable/paginated) ===== */}
   189→      <ApplicationProgressSection apps={apps} onViewAll={() => navigate("pm-applications")} onOpen={(id) => openApplication(id, "pm-application-details")} />
   190→
   191→      {/* ===== Live Workflow Monitor + Recent Activity (SIDE-BY-SIDE on desktop) ===== */}
   192→      <div className="grid grid-cols-1 gap-5 lg:grid-cols-[minmax(0,1fr)_minmax(0,1fr)]">
   193→        <LiveWorkflowSection apps={apps} onOpen={(id) => openApplication(id, "pm-application-details")} onViewAll={() => navigate("pm-workflow")} />
   194→        <RecentActivitySection apps={apps} onViewAll={() => navigate("pm-reports")} />
   195→      </div>
   196→
   197→      {/* ===== SLA Summary + Current Bottleneck (2-col) ===== */}
   198→      <div className="grid grid-cols-1 gap-5 lg:grid-cols-2">
   199→        <SlaSummarySection slaSummary={slaSummary} onCategoryClick={(status) => navigate("pm-sla")} />
   200→        <BottleneckSection bottleneck={bottleneck} onInspect={() => navigate("pm-sla")} />
   201→      </div>
   202→
   203→      {/* ===== Officer Workload (full width, compact TABLE) ===== */}
   204→      <OfficerWorkloadSection apps={apps} users={users} onViewAll={() => navigate("pm-officers")} onOpenOfficer={(id) => openApplication(id, "pm-officer-details")} />
   205→
   206→      {/* ===== Pending Actions (full width, searchable/filterable/paginated) ===== */}
   207→      <PendingActionsSection apps={apps} onOpen={(id) => openApplication(id, "pm-application-details")} onViewAll={() => navigate("pm-applications")} />
   208→    </div>
   209→  );
   210→}

   255→// ============================================================
   256→// APPLICATION PROGRESS SECTION
   257→// ============================================================
   258→function ApplicationProgressSection({
   259→  apps,
   260→  onViewAll,
   261→  onOpen,
   262→}: {
   263→  apps: Application[];
   264→  onViewAll: () => void;
   265→  onOpen: (id: string) => void;
   266→}) {
   267→  const [query, setQuery] = React.useState("");
   268→  const [statusFilter, setStatusFilter] = React.useState("ALL");
   269→  const [stageFilter, setStageFilter] = React.useState("ALL");
   270→  const [roleFilter, setRoleFilter] = React.useState("ALL");
   271→  const [slaFilter, setSlaFilter] = React.useState("ALL");
   272→  const { page, setPage, reset } = usePmPagination();
   273→  const pageSize = 10;
   274→
   275→  const filtered = React.useMemo(() => {
   276→    let list = [...apps].sort((a, b) => b.lastUpdated.localeCompare(a.lastUpdated));
   277→    if (query.trim()) {
   278→      const q = query.toLowerCase();
   279→      list = list.filter((a) =>
   280→        a.applicationNo.toLowerCase().includes(q) ||
   281→        a.project.name.toLowerCase().includes(q) ||
   282→        a.applicant.name.toLowerCase().includes(q) ||
   283→        (a.assignedOfficer?.name ?? "").toLowerCase().includes(q) ||
   284→        a.currentStageLabel.toLowerCase().includes(q) ||
   285→        a.status.toLowerCase().includes(q)
   286→      );
   287→    }
   288→    if (statusFilter !== "ALL") list = list.filter((a) => a.status === statusFilter);
   289→    if (stageFilter !== "ALL") list = list.filter((a) => a.currentStage === stageFilter);
   290→    if (roleFilter !== "ALL") {
   291→      list = list.filter((a) => {
   292→        const roles = rolesForStage(a.currentStage);
   293→        return roles.includes(roleFilter as RoleKey) || a.assignedOfficer?.role === roleFilter;
   294→      });
   295→    }
   296→    if (slaFilter !== "ALL") list = list.filter((a) => computeSLA(a).status === slaFilter);
   297→    return list;
   298→  }, [apps, query, statusFilter, stageFilter, roleFilter, slaFilter]);
   299→
   300→  // Reset to page 1 when filters change
   301→  React.useEffect(() => { reset(); }, [query, statusFilter, stageFilter, roleFilter, slaFilter, reset]);
   302→
   303→  const pageApps = filtered.slice((page - 1) * pageSize, page * pageSize);
   304→
   305→  return (
   306→    <PmSection>
   307→      <PmCardHeader
   308→        icon={Activity}
   309→        title="Application Progress Overview"
   310→        subtitle="Latest applications across the approval workflow"
   311→        controls={
   312→          <div className="flex items-center gap-2 flex-wrap">
   313→            <PmSearchInput value={query} onChange={setQuery} placeholder="Search applications…" className="w-full sm:w-48" />
   314→            <PmFilterSelect value={statusFilter} onChange={setStatusFilter} options={STATUS_OPTIONS} ariaLabel="Filter by status" className="w-full sm:w-36" />
   315→            <PmFilterSelect value={stageFilter} onChange={setStageFilter} options={STAGE_OPTIONS} ariaLabel="Filter by stage" className="w-full sm:w-36" />
   316→            <PmFilterSelect value={roleFilter} onChange={setRoleFilter} options={ROLE_OPTIONS} ariaLabel="Filter by role" className="w-full sm:w-32" />
   317→            <PmFilterSelect value={slaFilter} onChange={setSlaFilter} options={SLA_OPTIONS} ariaLabel="Filter by SLA" className="w-full sm:w-32" />
   318→          </div>
   319→        }
   320→      />
   321→      {pageApps.length === 0 ? (
   322→        <PmEmptyState
   323→          message="No applications match your filters."
   324→          onClear={() => { setQuery(""); setStatusFilter("ALL"); setStageFilter("ALL"); setRoleFilter("ALL"); setSlaFilter("ALL"); }}
   325→        />
   326→      ) : (
   327→        <div className="overflow-x-auto">
   328→          <table className="w-full text-sm">
   329→            <thead className="bg-muted/40">
   330→              <tr className="border-b-2 border-border text-left text-[10px] uppercase tracking-wide text-foreground">
   331→                <th scope="col" className="px-4 py-2.5 font-bold w-[12%]">Application No.</th>
   332→                <th scope="col" className="px-4 py-2.5 font-bold w-[16%]">Project</th>
   333→                <th scope="col" className="px-4 py-2.5 font-bold w-[12%]">Applicant</th>
   334→                <th scope="col" className="px-4 py-2.5 font-bold w-[12%]">Current Stage</th>
   335→                <th scope="col" className="px-4 py-2.5 font-bold w-[8%]">Role</th>
   336→                <th scope="col" className="px-4 py-2.5 font-bold w-[12%]">Officer</th>
   337→                <th scope="col" className="px-4 py-2.5 font-bold w-[10%]">Status</th>
   338→                <th scope="col" className="px-4 py-2.5 text-right font-bold w-[8%]">Progress</th>
   339→                <th scope="col" className="px-4 py-2.5 font-bold w-[5%]">SLA</th>
   340→                <th scope="col" className="px-4 py-2.5 font-bold w-[5%]">Updated</th>
   341→              </tr>
   342→            </thead>
   343→            <tbody className="divide-y divide-border">
   344→              {pageApps.map((a) => {
   345→                const sla = computeSLA(a);
   346→                return (
   347→                  <tr
   348→                    key={a.id}
   349→                    onClick={() => onOpen(a.id)}
   350→                    className="cursor-pointer transition-colors hover:bg-muted/30"
   351→                    style={{ height: "52px" }}
   352→                  >
   353→                    <td className="px-4 py-2">
   354→                      <button
   355→                        onClick={(e) => { e.stopPropagation(); onOpen(a.id); }}
   356→                        className="font-mono text-xs font-medium text-primary hover:underline"
   357→                        title={a.applicationNo}
   358→                      >
   359→                        {a.applicationNo}
   360→                      </button>
   361→                    </td>
   362→                    <td className="px-4 py-2 max-w-[180px]">
   363→                      <p className="truncate text-xs font-medium" title={a.project.name}>{a.project.name}</p>
   364→                    </td>
   365→                    <td className="px-4 py-2 max-w-[140px]">
   366→                      <p className="truncate text-xs" title={a.applicant.name}>{a.applicant.name}</p>
   367→                    </td>
   368→                    <td className="px-4 py-2">
   369→                      <span className="text-xs truncate" title={a.currentStageLabel}>{a.currentStageLabel}</span>
   370→                    </td>
   371→                    <td className="px-4 py-2">
   372→                      <span className="text-xs text-muted-foreground">
   373→                        {a.assignedOfficer ? a.assignedOfficer.role.replace("_", " ") : "—"}
   374→                      </span>
   375→                    </td>
   376→                    <td className="px-4 py-2 max-w-[140px]">
   377→                      <p className="truncate text-xs" title={a.assignedOfficer?.name}>
   378→                        {a.assignedOfficer?.name ?? "—"}
   379→                      </p>
   380→                    </td>
   381→                    <td className="px-4 py-2"><StatusBadge status={a.status} showIcon={false} /></td>
   382→                    <td className="px-4 py-2">
   383→                      <div className="flex items-center gap-2 justify-end">
   384→                        <Progress value={a.progress} className="h-1.5 w-12" />
   385→                        <span className="text-xs tabular-nums w-8 text-right">{a.progress}%</span>
   386→                      </div>
   387→                    </td>
   388→                    <td className="px-4 py-2">
   389→                      <Badge variant="outline" className={cn("text-[9px] whitespace-nowrap", sla.cls)}>{sla.label}</Badge>
   390→                    </td>
   391→                    <td className="px-4 py-2 text-xs text-muted-foreground whitespace-nowrap">{timeAgoBrief(a.lastUpdated)}</td>
   392→                  </tr>
   393→                );
   394→              })}
   395→            </tbody>
   396→          </table>
   397→        </div>
   398→      )}
   399→      <div className="flex items-center justify-between border-t border-border px-4 py-2.5">
   400→        <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={onViewAll}>
   401→          View All <ArrowRight className="size-3" />
   402→        </Button>
   403→      </div>
   404→      <PmPagination page={page} pageSize={pageSize} total={filtered.length} onPageChange={setPage} />
   405→    </PmSection>
   406→  );
   407→}

   238→   238→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   239→   239→  const newStage = stage ?? stageFromStatus(status);
   240→   240→  const stageInfo = getStage(newStage);
   241→   241→  return {
   242→   242→    ...app,
   243→   243→    status,
   244→   244→    currentStage: newStage,
   245→   245→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   246→   246→    lastUpdated: nowISO(),

   248→   248→  };
   249→   249→}

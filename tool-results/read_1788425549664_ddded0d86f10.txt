    50→// PROJECT MANAGER DASHBOARD
    51→// Central operational view — READ-ONLY monitoring.
    52→// No edit/approve/reject/verify/pay buttons anywhere.
    53→// ============================================================
    54→
    55→export function PmDashboard() {
    56→  const navigate = useAppStore((s) => s.navigate);
    57→  const openApplication = useAppStore((s) => s.openApplication);
    58→
    59→  const apps = useAllApplications();
    60→  const users = useAllUsers();
    61→
    62→  // ---- KPI counts derived from the shared applications dataset ----
    63→  const kpis = React.useMemo(() => {
    64→    const total = apps.length;
    65→    const inProgress = apps.filter(
    66→      (a) =>
    67→        !["APPROVED", "REJECTED", "DRAFT"].includes(a.status)
    68→    ).length;
    69→    const approved = apps.filter((a) => a.status === "APPROVED").length;

    71→      const sla = computeSLA(a);
    72→      return (
    73→        sla.status === "DELAYED" ||
    74→        sla.status === "CRITICAL" ||
    75→        sla.status === "AT_RISK" ||
    76→        sla.status === "BLOCKED"
    77→      );
    78→    }).length;
    79→    return { total, inProgress, approved, delayedAtRisk };
    80→  }, [apps]);
    81→
    82→  // ---- Application Progress Overview (max 10 rows) ----
    83→  const overviewApps = React.useMemo(
    84→    () =>
    85→      [...apps]
    86→        .sort((a, b) => b.lastUpdated.localeCompare(a.lastUpdated))
    87→        .slice(0, 10),
    88→    [apps]
    89→  );
    90→
    91→  // ---- Live Workflow Monitor: 5 most-recently-updated non-completed apps ----
    92→  const liveApps = React.useMemo(
    93→    () =>
    94→      apps
    95→        .filter((a) => !["APPROVED", "REJECTED", "DRAFT"].includes(a.status))
    96→        .sort((a, b) => b.lastUpdated.localeCompare(a.lastUpdated))
    97→        .slice(0, 5),
    98→    [apps]
    99→  );
   100→
   101→  // ---- SLA Summary counts (across all apps) ----
   102→  const slaSummary = React.useMemo(() => {
   103→    const counts: Record<
   104→      "ON_TRACK" | "AT_RISK" | "DELAYED" | "CRITICAL" | "BLOCKED",
   105→      number
   106→    > = {
   107→      ON_TRACK: 0,
   108→      AT_RISK: 0,
   109→      DELAYED: 0,
   110→      CRITICAL: 0,
   111→      BLOCKED: 0,
   112→    };

   114→      const sla = computeSLA(a);
   115→      if (sla.status === "COMPLETED") return;

   119→  }, [apps]);
   120→
   121→  // ---- Officer workload ----
   122→  const officerWorkloads = React.useMemo(
   123→    () => computeOfficerWorkloads(apps, users).slice(0, 6),

   125→  );
   126→

   129→
   130→  // ---- Pending Actions (top 5) ----

   132→    () => computePendingActions(apps).slice(0, 5),
   133→    [apps]
   134→  );
   135→
   136→  // ---- Recent Activity (15 events) ----
   137→  const recentActivity = React.useMemo(
   138→    () => computeRecentActivity(apps, 15),
   139→    [apps]
   140→  );
   141→
   142→  return (
   143→    <div className="space-y-6">
   144→      <PageHeader
   145→        title="Project Manager Dashboard"
   146→        description="Central operational view of the building permit approval workflow."
   147→        icon={BarChart3}
   148→        breadcrumbs={[
   149→          { label: "Project Manager" },
   150→          { label: "Dashboard" },
   151→        ]}
   152→      />
   153→
   154→      {/* ===== KPI CARDS (4-col grid) ===== */}
   155→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   156→        <StatCard
   157→          label="Total Applications"
   158→          value={kpis.total}
   159→          icon={FileStack}
   160→          accent="primary"
   161→          onClick={() => navigate("pm-applications")}
   162→        />
   163→        <StatCard
   164→          label="In Progress"
   165→          value={kpis.inProgress}
   166→          icon={Clock}
   167→          accent="info"
   168→          onClick={() => navigate("pm-applications")}
   169→        />
   170→        <StatCard
   171→          label="Approved"
   172→          value={kpis.approved}
   173→          icon={CheckCircle2}
   174→          accent="success"
   175→          onClick={() => navigate("pm-applications")}
   176→        />
   177→        <StatCard
   178→          label="Delayed / At Risk"
   179→          value={kpis.delayedAtRisk}
   180→          icon={AlertTriangle}
   181→          accent="destructive"
   182→          onClick={() => navigate("pm-sla")}
   183→        />
   184→      </div>
   185→
   186→      {/* ===== Application Progress Overview (full-width table) ===== */}
   187→      <SectionCard
   188→        title="Application Progress Overview"
   189→        description="Latest applications across the workflow"
   190→        icon={Activity}
   191→        noPadding
   192→        action={
   193→          <Button
   194→            variant="outline"
   195→            size="sm"
   196→            className="h-8 text-xs"
   197→            onClick={() => navigate("pm-applications")}
   198→          >
   199→            View all <ArrowRight className="size-3" />
   200→          </Button>
   201→        }
   202→      >
   203→        {overviewApps.length === 0 ? (
   204→          <div className="p-6">
   205→            <EmptyState
   206→              icon={FileStack}
   207→              title="No applications found"
   208→              description="Applications will appear here once they are created."
   209→            />

   211→        ) : (
   212→          <div className="overflow-x-auto">
   213→            <table className="w-full text-sm">
   214→              <caption className="sr-only">
   215→                Application progress overview
   216→              </caption>
   217→              <thead className="sticky top-0 z-10 bg-muted/60 backdrop-blur">
   218→                <tr className="border-b-2 border-border text-left text-[11px] uppercase tracking-wide text-foreground">
   219→                  <th scope="col" className="px-4 py-3 font-bold">
   220→                    Application No.
   221→                  </th>
   222→                  <th scope="col" className="px-4 py-3 font-bold">
   223→                    Project
   224→                  </th>
   225→                  <th scope="col" className="px-4 py-3 font-bold">
   226→                    Applicant
   227→                  </th>
   228→                  <th scope="col" className="px-4 py-3 font-bold">
   229→                    Current Stage
   230→                  </th>
   231→                  <th scope="col" className="px-4 py-3 font-bold">
   232→                    Assigned Role
   233→                  </th>
   234→                  <th scope="col" className="px-4 py-3 font-bold">

   270→                          {a.applicationNo}
   271→                        </button>

   274→                        <p className="truncate text-xs font-medium">
   275→                          {a.project.name}
   276→                        </p>
   277→                      </td>
   278→                      <td className="px-4 py-3 max-w-[160px]">
   279→                        <p className="truncate text-xs">{a.applicant.name}</p>
   280→                      </td>
   281→                      <td className="px-4 py-3">
   282→                        <span className="text-xs">{a.currentStageLabel}</span>
   283→                      </td>
   284→                      <td className="px-4 py-3">
   285→                        {a.assignedOfficer ? (
   286→                          <RoleBadge role={a.assignedOfficer.role} />
   287→                        ) : (
   288→                          <span className="text-xs text-muted-foreground">—</span>
   289→                        )}
   290→                      </td>
   291→                      <td className="px-4 py-3">
   292→                        <span className="text-xs">
   293→                          {a.assignedOfficer?.name ?? "Unassigned"}
   294→                        </span>
   295→                      </td>
   296→                      <td className="px-4 py-3">
   297→                        <StatusBadge status={a.status} showIcon={false} />
   298→                      </td>
   299→                      <td className="px-4 py-3 text-right">
   300→                        <div className="flex items-center gap-2 justify-end">
   301→                          <Progress
   302→                            value={a.progress}
   303→                            className="h-1.5 w-16"
   304→                          />
   305→                          <span className="text-[11px] tabular-nums text-muted-foreground">
   306→                            {a.progress}%
   307→                          </span>
   308→                        </div>
   309→                      </td>
   310→                      <td className="px-4 py-3">
   311→                        <SlaBadge label={sla.label} cls={sla.cls} />
   312→                      </td>
   313→                      <td className="px-4 py-3 text-xs text-muted-foreground whitespace-nowrap">
   314→                        {timeAgoBrief(a.lastUpdated)}
   315→                      </td>
   316→                    </tr>
   317→                  );
   318→                })}
   319→              </tbody>
   320→            </table>
   321→          </div>
   322→        )}
   323→      </SectionCard>
   324→
   325→      {/* ===== ROW: Live Workflow Monitor (LEFT, span-2) + SLA Summary (RIGHT) ===== */}
   326→      <div className="grid grid-cols-1 gap-6 xl:grid-cols-[1fr_320px]">
   327→        <SectionCard
   328→          title="Live Workflow Monitor"
   329→          description="Most recently updated active applications"
   330→          icon={Activity}
   331→          className="xl:col-span-2"
   332→        >
   333→          {liveApps.length === 0 ? (
   334→            <EmptyState
   335→              icon={CircleDot}
   336→              title="No active applications"
   337→              description="All applications are currently completed or in draft."
   338→            />
   339→          ) : (
   340→            <ul className="space-y-2.5">
   341→              {liveApps.map((a) => {
   342→                const sla = computeSLA(a);
   343→                return (
   344→                  <li key={a.id}>
   345→                    <button
   346→                      onClick={() =>
   347→                        openApplication(a.id, "pm-application-details")
   348→                      }
   349→                      className="flex w-full items-center gap-3 rounded-lg border border-border bg-card p-3 text-left transition-colors hover:border-primary/40 hover:bg-muted/30"
   350→                    >
   351→                      <div className="min-w-0 flex-1 space-y-1">
   352→                        <div className="flex items-center gap-2 flex-wrap">
   353→                          <span className="font-mono text-xs font-medium text-primary">
   354→                            {a.applicationNo}
   355→                          </span>
   356→                          <span className="truncate text-xs font-medium">
   357→                            {a.project.name}
   358→                          </span>
   359→                        </div>
   360→                        <div className="flex flex-wrap items-center gap-2 text-[11px] text-muted-foreground">
   361→                          <span>{a.currentStageLabel}</span>
   362→                          <span aria-hidden>·</span>
   363→                          <span>
   364→                            {a.assignedOfficer?.name ?? "Unassigned"}
   365→                          </span>
   366→                          {a.assignedOfficer && (
   367→                            <RoleBadge role={a.assignedOfficer.role} />
   368→                          )}
   369→                          <span aria-hidden>·</span>
   370→                          <span>
   371→                            Pending since {timeAgoBrief(a.lastUpdated)}
   372→                          </span>
   373→                        </div>
   374→                      </div>
   375→                      <div className="flex flex-col items-end gap-1.5 shrink-0">
   376→                        <div className="flex items-center gap-2">
   377→                          <Progress
   378→                            value={a.progress}
   379→                            className="h-1.5 w-20"
   380→                          />
   381→                          <span className="text-[11px] tabular-nums text-muted-foreground">
   382→                            {a.progress}%
   383→                          </span>
   384→                        </div>
   385→                        <SlaBadge
   386→                          label={sla.label}
   387→                          cls={sla.cls}
   388→                        />
   389→                      </div>
   390→                    </button>
   391→                  </li>
   392→                );
   393→              })}
   394→            </ul>
   395→          )}
   396→        </SectionCard>
   397→
   398→        <SectionCard
   399→          title="SLA Summary"
   400→          description="Distribution of SLA statuses across all applications"
   401→          icon={Gauge}
   402→        >
   403→          <ul className="space-y-2">
   404→            {SLA_SUMMARY_ITEMS.map((item) => (
   405→              <li key={item.key}>
   406→                <button
   407→                  onClick={() => navigate("pm-sla")}
   408→                  className="flex w-full items-center justify-between gap-2 rounded-lg border border-border bg-card p-3 transition-colors hover:border-primary/40 hover:bg-muted/30"
   409→                  aria-label={`${item.label}: ${slaSummary[item.key]} applications. View SLA details.`}
   410→                >
   411→                  <div className="flex items-center gap-2 min-w-0">
   412→                    <span
   413→                      className={cn(
   414→                        "flex size-8 shrink-0 items-center justify-center rounded-lg",
   415→                        item.cls
   416→                      )}
   417→                    >
   418→                      <item.icon className="size-4" />
   419→                    </span>
   420→                    <span className="text-xs font-medium">{item.label}</span>
   421→                  </div>
   422→                  <div className="flex items-center gap-2 shrink-0">
   423→                    <Badge
   424→                      variant="outline"
   425→                      className="tabular-nums"
   426→                    >
   427→                      {slaSummary[item.key]}
   428→                    </Badge>
   429→                    <ChevronRight className="size-3.5 text-muted-foreground" />
   430→                  </div>
   431→                </button>
   432→              </li>
   433→            ))}
   434→          </ul>
   435→        </SectionCard>
   436→      </div>
   437→
   438→      {/* ===== ROW: Officer Workload (LEFT) + Current Bottleneck (RIGHT) ===== */}
   439→      <div className="grid grid-cols-1 gap-6 xl:grid-cols-[1fr_320px]">
   440→        <SectionCard
   441→          title="Officer Workload"
   442→          description="Active applications assigned per officer"
   443→          icon={Users}
   444→          className="xl:col-span-2"
   445→          action={
   446→            <Button
   447→              variant="outline"
   448→              size="sm"
   449→              className="h-8 text-xs"
   450→              onClick={() => navigate("pm-officers")}
   451→            >
   452→              View all <ArrowRight className="size-3" />

   459→              title="No officers found"
   460→              description="There are no active officers in the system yet."
   461→            />
   462→          ) : (
   463→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   464→              {officerWorkloads.map((w) => {
   465→                const maxAssigned = Math.max(
   466→                  1,

   470→                  maxAssigned === 0 ? 0 : (w.assigned / maxAssigned) * 100;

   166→   166→function updateApp(
   167→   167→  apps: Application[],
   168→   168→  appId: string,
   169→   169→  updater: (app: Application) => Application
   170→   170→): Application[] {
   171→   171→  return apps.map((a) => (a.id === appId ? updater(a) : a));

   174→   174→function addAudit(app: Application, entry: Omit<AuditEntry, "id" | "timestamp" | "entity" | "entityId">): Application {
   175→   175→  const audit: AuditEntry = {
   176→   176→    ...entry,
   177→   177→    id: genId("audit"),
   178→   178→    entity: "Application",
   179→   179→    entityId: app.applicationNo,
   180→   180→    timestamp: nowISO(),
   181→   181→  };
   182→   182→  return { ...app, auditLog: [...app.auditLog, audit] };
   183→   183→}

   186→   186→  app: Application,
   187→   187→  stage: WorkflowStageKey,
   188→   188→  actor: { name: string; role: RoleKey },
   189→   189→  action: string,

   199→   199→    remarks,
   200→   200→    timestamp: nowISO(),
   201→   201→    status,
   202→   202→  };
   203→   203→  return { ...app, workflowHistory: [...app.workflowHistory, entry] };
   204→   204→}

   206→   206→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   207→   207→  const newStage = stage ?? stageFromStatus(status);
   208→   208→  const stageInfo = getStage(newStage);
   209→   209→  return {
   210→   210→    ...app,
   211→   211→    status,
   212→   212→    currentStage: newStage,
   213→   213→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   214→   214→    lastUpdated: nowISO(),
   215→   215→    progress: computeProgress(status, newStage),
   216→   216→  };
   217→   217→}

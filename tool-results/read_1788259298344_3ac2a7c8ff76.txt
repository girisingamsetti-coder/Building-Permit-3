   166→function updateApp(
   167→  apps: Application[],
   168→  appId: string,
   169→  updater: (app: Application) => Application
   170→): Application[] {
   171→  return apps.map((a) => (a.id === appId ? updater(a) : a));
   172→}

   174→function addAudit(app: Application, entry: Omit<AuditEntry, "id" | "timestamp" | "entity" | "entityId">): Application {
   175→  const audit: AuditEntry = {
   176→    ...entry,
   177→    id: genId("audit"),
   178→    entity: "Application",
   179→    entityId: app.applicationNo,
   180→    timestamp: nowISO(),
   181→  };
   182→  return { ...app, auditLog: [...app.auditLog, audit] };
   183→}
   184→
   185→function addWorkflowHistory(
   186→  app: Application,
   187→  stage: WorkflowStageKey,
   188→  actor: { name: string; role: RoleKey },
   189→  action: string,
   190→  remarks?: string,
   191→  status: WorkflowHistoryEntry["status"] = "COMPLETED"
   192→): Application {
   193→  const entry: WorkflowHistoryEntry = {
   194→    id: genId("wf"),
   195→    stage,
   196→    stageLabel: getStage(stage)?.label ?? stage,
   197→    actor,
   198→    action,
   199→    remarks,
   200→    timestamp: nowISO(),
   201→    status,
   202→  };
   203→  return { ...app, workflowHistory: [...app.workflowHistory, entry] };
   204→}
   205→
   206→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   207→  const newStage = stage ?? stageFromStatus(status);
   208→  const stageInfo = getStage(newStage);
   209→  return {
   210→    ...app,
   211→    status,
   212→    currentStage: newStage,
   213→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   214→    lastUpdated: nowISO(),
   215→    progress: computeProgress(status, newStage),
   216→  };
   217→}

   430→  runScrutiny: (appId) => {

   437→        const latest = app.drawings[app.drawings.length - 1];
   438→        let updated: Application = { ...app, drawings: app.drawings.map((d) => d.id === latest.id ? { ...d, status: "SCRUTINY_IN_PROGRESS" as const } : d) };
   439→        updated = addAudit(updated, { user: "System", role: "TPS", action: `Auto-scrutiny started (v${latest.version})`, oldStatus: app.status, newStatus: "SCRUTINY_IN_PROGRESS" });

   449→          // Deterministic: v1 fails (front setback), v2+ passes with warnings (so LTP experiences fail→reupload→pass)
   450→          const scenario = latest.version >= 2 ? "passed_warnings" : "front_setback";

   453→          let updated: Application = { ...app, scrutinyReport: report };
   454→          updated = { ...updated, drawings: updated.drawings.map((d) => d.id === latest.id ? { ...d, status: passed ? "SCRUTINY_PASSED" as const : "SCRUTINY_FAILED" as const } : d) };
   455→          const newStatus = passed ? "SCRUTINY_PASSED" : "SCRUTINY_FAILED";
   456→          updated = addAudit(updated, { user: "System", role: "TPS", action: `Auto-scrutiny ${passed ? "passed" : "failed"} (v${latest.version})`, oldStatus: app.status, newStatus });

   460→            updated = addWorkflowHistory(updated, "DRAWING_SCRUTINY", { name: "System (Auto-Scrutiny)", role: "TPS" }, "Scrutiny failed — re-upload required", report.summary, "FAILED");
   461→          }
   462→          updated = setAppStatus(updated, newStatus, passed ? "DOCUMENTS" : "DRAWING_SCRUTINY");
   463→          return updated;
   464→        }),
   465→        processingAppIds: s.processingAppIds.filter((id) => id !== appId),
   466→      }));
   467→      // Side effects: notification + SMS
   468→      const app = get().applications.find((a) => a.id === appId)!;

   471→        : NotificationFactory.scrutinyFailed(app);
   472→      set((s) => ({
   473→        notifications: [notification, ...s.notifications],
   474→        smsLogs: smsLog ? [smsLog, ...s.smsLogs] : s.smsLogs,
   475→      }));
   476→    }, 2500);
   477→  },

   487→        updated = { ...updated, auditLog: [...updated.auditLog, audit] };
   488→        updated = setAppStatus(updated, "DRAWING_UPLOADED", "DRAWING_SCRUTINY");
   489→        return updated;
   490→      }),
   491→    }));
   492→  },

   494→  uploadDocument: (appId, docCode, fileName, fileSize) => {
   495→    const user = get().user!;
   496→    set((s) => ({
   497→      applications: updateApp(s.applications, appId, (app) => {
   498→        let updated: Application = { ...app, documents: app.documents.map((d) => d.code === docCode ? { ...d, status: "UPLOADED" as const, uploadedAt: nowISO(), version: (d.version ?? 0) + 1, fileSize } : d) };
   499→        // If all required docs uploaded, move to DOCUMENT_VERIFICATION

   861→        return updated;
   862→      }),
   863→    }));
   864→    // Notification if fee generated
   865→    const app = get().applications.find((a) => a.id === appId)!;
   866→    if (app.status === "FEE_GENERATED" && app.fee) {
   867→      const { notification, smsLog } = NotificationFactory.feeGenerated(app);
   868→      set((s) => ({ notifications: [notification, ...s.notifications], smsLogs: smsLog ? [smsLog, ...s.smsLogs] : s.smsLogs }));
   869→    }
   870→  },
   871→
   872→  rejectDocument: (appId, docId, reason) => {
   873→    const user = get().user!;
   874→    set((s) => ({
   875→      applications: updateApp(s.applications, appId, (app) => {
   876→        let updated: Application = { ...app, documents: app.documents.map((d) => d.id === docId ? { ...d, status: "REJECTED" as const, verifiedBy: user.name, verifiedAt: nowISO(), remarks: reason } : d) };
   877→        updated = addAudit(updated, { user: user.name, role: user.role, action: `Document rejected: ${updated.documents.find((d) => d.id === docId)?.name}`, remarks: reason });
   878→        return updated;
   879→      }),
   880→    }));
   881→  },
   882→
   883→  addRemark: (appId, text, type) => {
   884→    const user = get().user!;
   885→    set((s) => ({
   886→      applications: updateApp(s.applications, appId, (app) => {
   887→        const remark: Remark = { id: genId("rem"), author: { name: user.name, role: user.role }, text, timestamp: nowISO(), type };
   888→        let updated: Application = { ...app, remarks: [...app.remarks, remark] };
   889→        updated = addAudit(updated, { user: user.name, role: user.role, action: `Remark added (${type})`, remarks: text });
   890→        return updated;
   891→      }),
   892→    }));
   893→  },
   894→
   895→  // ---- NOTIFICATIONS ----
   896→  markNotificationRead: (id) =>
   897→    set((s) => ({ notifications: s.notifications.map((n) => n.id === id ? { ...n, read: true } : n) })),
   898→  markAllNotificationsRead: () =>
   899→    set((s) => ({ notifications: s.notifications.map((n) => ({ ...n, read: true })) })),
   900→}));

   909→export function useVisibleApplications() {
   910→  const applications = useAppStore((s) => s.applications);
   911→  const user = useAppStore((s) => s.user);
   912→  return useMemo(() => {
   913→    if (!user) return [];
   914→    if (user.role === "ADMIN") return applications;
   915→    if (user.role === "LTP") return applications.filter((a) => a.ltpId === user.id);
   916→    return applications.filter((a) => {
   917→      const roles = rolesForStage(a.currentStage);
   918→      if (roles.includes(user.role) && !["APPROVED", "REJECTED"].includes(a.status)) return true;
   919→      return a.workflowHistory.some((w) => w.actor.role === user.role);
   920→    });
   921→  }, [applications, user]);
   922→}

   924→export function useAssignedApplications() {
   925→  const applications = useAppStore((s) => s.applications);
   926→  const user = useAppStore((s) => s.user);

   929→    return applications.filter((a) => {

   931→      return rolesForStage(a.currentStage).includes(user.role);
   932→    });
   933→  }, [applications, user]);

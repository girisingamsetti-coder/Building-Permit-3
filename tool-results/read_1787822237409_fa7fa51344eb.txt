   199→type PayStage = "pending" | "method" | "processing" | "success" | "failed";
   200→
   201→export function LtpPayment() {
   202→  const { navigate, applications } = useAppStore();
   203→  const app = useAppOrDefault();
   204→  const { toast } = useToast();
   205→  const [method, setMethod] = React.useState("UPI");
   206→  const [stage, setStage] = React.useState<PayStage>(app?.payment?.status === "SUCCESSFUL" ? "success" : "pending");
   207→  const appId = app?.id;
   208→  React.useEffect(() => {
   209→    setStage(app?.payment?.status === "SUCCESSFUL" ? "success" : "pending");
   210→    setMethod("UPI");
   211→  }, [appId, app?.payment?.status]);
   212→
   213→  if (!app || !app.fee) {
   214→    return (
   215→      <div className="space-y-6">
   216→        <PageHeader title="Payment" icon={CreditCard} breadcrumbs={[{ label: "LTP Portal", onClick: () => navigate("ltp-dashboard") }, { label: "Payment" }]} />
   217→        <EmptyState icon={CreditCard} title="No payment due" description="There is no outstanding payment for this application." />
   218→      </div>
   219→    );
   220→  }
   221→
   222→  function initiatePayment() {
   223→    setStage("processing");
   224→    setTimeout(() => {
   225→      setStage("success");
   226→      toast({ title: "Payment Successful", description: "Your payment has been verified and receipt generated." });
   227→    }, 2600);
   228→  }
   229→
   230→  const PAYMENT_STEPS = [
   231→    { key: "pending", label: "Fee Pending" },
   232→    { key: "method", label: "Select Method" },
   233→    { key: "processing", label: "Processing" },
   234→    { key: "success", label: "Successful" },
   235→  ];
   236→
   237→  return (
   238→    <div className="space-y-6">
   239→      <PageHeader
   240→        title="Payment"
   241→        description="Secure online payment for your application fees."
   242→        icon={CreditCard}
   243→        breadcrumbs={[{ label: "LTP Portal", onClick: () => navigate("ltp-dashboard") }, { label: "Payment" }]}
   244→        actions={<AppSelect apps={applications} current={app} view="ltp-payment" />}
   245→      />
   246→
   247→      {/* Payment progress stepper */}
   248→      <div className="rounded-xl border border-border bg-card p-4 shadow-gov">
   249→        <ol className="flex items-center justify-between">
   250→          {PAYMENT_STEPS.map((s, idx) => {
   251→            const currentIdx = PAYMENT_STEPS.findIndex((p) => p.key === (stage === "failed" ? "processing" : stage));
   252→            const isActive = idx === currentIdx;
   253→            const isDone = idx < currentIdx;

   255→              <li key={s.key} className="flex flex-1 items-center">

   258→                    {isDone ? <Check className="size-4" /> : idx + 1}
   259→                  </div>
   260→                  <span className={cn("hidden text-xs font-medium sm:block", isActive ? "text-foreground" : "text-muted-foreground")}>{s.label}</span>
   261→                </div>
   262→                {idx < PAYMENT_STEPS.length - 1 && <div className={cn("mx-2 h-0.5 flex-1 rounded-full", isDone ? "bg-success" : "bg-border")} />}

   269→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-3">
   270→        <div className="lg:col-span-2 space-y-6">

   276→                    <p className="text-xs text-muted-foreground">Application</p>

   279→                  <StatusBadge status={app.status} showIcon={false} />

   283→                    <span className="text-muted-foreground">{li.name}</span>

   403→function ProcessingStep({ label, done, active }: { label: string; done?: boolean; active?: boolean }) {

   408→    </div>
   409→  );
   410→}

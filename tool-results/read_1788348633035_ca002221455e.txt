   267→type PayStage = "pending" | "method" | "processing" | "success" | "failed";
   268→
   269→export function LtpPayment() {
   270→  const { navigate, openApplication, initiatePayment: storeInitiatePayment, selectedApplicationId } = useAppStore();
   271→  const visibleApps = useVisibleApplications();
   272→  const processingAppIds = useAppStore((s) => s.processingAppIds);
   273→  const { toast } = useToast();
   274→
   275→  // If a specific app is selected and it has a fee, show the payment flow for it
   276→  const selectedApp = selectedApplicationId
   277→    ? visibleApps.find((a) => a.id === selectedApplicationId)
   278→    : null;
   279→  const isPaymentFlow = selectedApp && selectedApp.fee;
   280→
   281→  // All apps with fees (for the payments listing)
   282→  const appsWithFees = visibleApps.filter((a) => a.fee);
   283→  const pendingPayments = appsWithFees.filter((a) =>
   284→    a.payment?.status === "PENDING" || a.status === "PAYMENT_PENDING" || a.status === "FEE_GENERATED"
   285→  );
   286→  const paidApps = appsWithFees.filter((a) => a.payment?.status === "SUCCESS");
   287→  const totalOutstanding = pendingPayments.reduce((sum, a) => sum + (a.fee?.outstanding ?? 0), 0);
   288→
   289→  // ===== Payment flow view (for a specific app) =====
   290→  if (isPaymentFlow && selectedApp) {
   291→    return <PaymentFlow app={selectedApp} navigate={navigate} storeInitiatePayment={storeInitiatePayment} processingAppIds={processingAppIds} toast={toast} />;
   292→  }
   293→
   294→  // ===== Payments listing view =====
   295→  return (
   296→    <div className="space-y-6">
   297→      <PageBackButton fallbackView="ltp-applications" fallbackLabel="Applications" />
   298→      <PageHeader
   299→        title="Payment"
   300→        description="Manage application fees, outstanding amounts and payment receipts."
   301→        icon={CreditCard}
   302→        breadcrumbs={[{ label: "LTP Portal", onClick: () => navigate("ltp-dashboard") }, { label: "Payment" }]}
   303→        badge={<Badge className="bg-warning/15 text-warning-foreground">Demo mode</Badge>}
   304→      />
   305→
   306→      {/* Summary Cards */}
   307→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   308→        <PaymentStatCard icon={Clock} value={pendingPayments.length} label="PENDING PAYMENTS" subtext="Applications awaiting payment" accent="bg-amber-500/10 text-amber-600" />
   309→        <PaymentStatCard icon={IndianRupee} value={formatINR(totalOutstanding)} label="TOTAL OUTSTANDING" subtext="Across pending applications" accent="bg-destructive/10 text-destructive" />
   310→        <PaymentStatCard icon={CheckCircle2} value={paidApps.length} label="PAID APPLICATIONS" subtext="Payment successfully completed" accent="bg-success/10 text-success" />

   313→
   314→      {/* Pending Payments */}
   315→      {pendingPayments.length > 0 && (
   316→        <SectionCard title="Pending Payments" description="Applications requiring fee payment" icon={Clock}>
   317→          <div className="space-y-3">
   318→            {pendingPayments.map((app) => (

   320→                <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   321→                  <div className="min-w-0 flex-1 space-y-1">
   322→                    <div className="flex items-center gap-2 flex-wrap">
   323→                      <p className="font-mono text-xs font-semibold text-primary">{app.applicationNo}</p>

   331→                  <div className="shrink-0 space-y-1 text-right">

    56→// ---------- Helpers ----------
    57→function daysRemaining(iso?: string): number | null {
    58→  if (!iso) return null;
    59→  const diff = new Date(iso).getTime() - Date.now();
    60→  return Math.ceil(diff / 86400000);
    61→}
    62→
    63→function slaTone(days: number | null): { cls: string; label: string; tone: "ok" | "near" | "critical" | "overdue" | "na" } {
    64→  if (days === null) return { cls: "text-muted-foreground", label: "—", tone: "na" };
    65→  if (days < 0) return { cls: "text-destructive font-semibold", label: `Overdue ${Math.abs(days)}d`, tone: "overdue" };
    66→  if (days <= 3) return { cls: "text-destructive font-semibold", label: `${days}d left`, tone: "critical" };
    67→  if (days <= 7) return { cls: "text-amber-600 dark:text-amber-400 font-semibold", label: `${days}d left`, tone: "near" };
    68→  return { cls: "text-success font-medium", label: `${days}d left`, tone: "ok" };
    69→}

   339→                  title="No SLA risks"

   569→// ---------- Helper: extract recent decisions for a role from store applications ----------
   570→function getRecentDecisions(
   571→  role: RoleKey,
   572→  apps: Application[]
   573→): { app: Application; entry: WorkflowHistoryEntry }[] {
   574→  const out: { app: Application; entry: WorkflowHistoryEntry }[] = [];
   575→  for (const app of apps) {
   576→    for (const entry of app.workflowHistory) {
   577→      if (entry.status !== "COMPLETED") continue;
   578→      const stage = getStage(entry.stage);
   579→      if (!stage) continue;
   580→      const stageRole = stage.role;
   581→      const isMatch =
   582→        role === stageRole ||
   583→        entry.actor.role === role ||
   584→        ((role === "TPS" || role === "TPA") && (stageRole === "TPS" || stageRole === "TPA")) ||
   585→        ((role === "ZAD" || role === "ZDD") && (stageRole === "ZAD" || stageRole === "ZDD"));
   586→      if (isMatch && entry.timestamp) {
   587→        out.push({ app, entry });
   588→      }
   589→    }
   590→  }
   591→  return out.sort((a, b) => b.entry.timestamp.localeCompare(a.entry.timestamp));
   592→}

   206→function updateApp(
   207→  apps: Application[],
   208→  appId: string,
   209→  updater: (app: Application) => Application
   210→): Application[] {
   211→  return apps.map((a) => (a.id === appId ? updater(a) : a));
   212→}

   214→function addAudit(app: Application, entry: Omit<AuditEntry, "id" | "timestamp" | "entity" | "entityId">): Application {
   215→  const audit: AuditEntry = {
   216→    ...entry,
   217→    id: genId("audit"),
   218→    entity: "Application",
   219→    entityId: app.applicationNo,
   220→    timestamp: nowISO(),
   221→  };
   222→  return { ...app, auditLog: [...app.auditLog, audit] };
   223→}
   224→
   225→function addWorkflowHistory(
   226→  app: Application,
   227→  stage: WorkflowStageKey,
   228→  actor: { name: string; role: RoleKey },
   229→  action: string,
   230→  remarks?: string,
   231→  status: WorkflowHistoryEntry["status"] = "COMPLETED"
   232→): Application {
   233→  const entry: WorkflowHistoryEntry = {
   234→    id: genId("wf"),
   235→    stage,
   236→    stageLabel: getStage(stage)?.label ?? stage,
   237→    actor,
   238→    action,
   239→    remarks,
   240→    timestamp: nowISO(),
   241→    status,
   242→  };
   243→  return { ...app, workflowHistory: [...app.workflowHistory, entry] };
   244→}
   245→
   246→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   247→  const newStage = stage ?? stageFromStatus(status);
   248→  const stageInfo = getStage(newStage);
   249→  return {
   250→    ...app,
   251→    status,
   252→    currentStage: newStage,
   253→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   254→    lastUpdated: nowISO(),
   255→    progress: computeProgress(status, newStage),
   256→  };
   257→}

  1360→export function useVisibleApplications() {
  1361→  const applications = useAppStore((s) => s.applications);
  1362→  const user = useAppStore((s) => s.user);
  1363→  return useMemo(() => {
  1364→    if (!user) return [];
  1365→    if (user.role === "ADMIN") return applications;
  1366→    if (user.role === "LTP") return applications.filter((a) => a.ltpId === user.id);
  1367→    return applications.filter((a) => {
  1368→      const roles = rolesForStage(a.currentStage);
  1369→      if (roles.includes(user.role) && !["APPROVED", "REJECTED"].includes(a.status)) return true;
  1370→      return a.workflowHistory.some((w) => w.actor.role === user.role);
  1371→    });
  1372→  }, [applications, user]);
  1373→}

  1396→    if (user.role === "ADMIN") return applications;
  1397→    if (user.role === "LTP") return applications.filter((a) => a.ltpId === user.id);
  1398→    // Officers with document:verify or document:reject see ALL applications that
  1399→    // have at least one uploaded document (so they can review documents even

  1401→    return applications.filter((a) =>

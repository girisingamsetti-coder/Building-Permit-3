   196→// INTERNAL HELPERS — update application immutably + side effects
   197→// ============================================================
   198→function updateApp(
   199→  apps: Application[],
   200→  appId: string,
   201→  updater: (app: Application) => Application
   202→): Application[] {
   203→  return apps.map((a) => (a.id === appId ? updater(a) : a));
   204→}
   205→
   206→function addAudit(app: Application, entry: Omit<AuditEntry, "id" | "timestamp" | "entity" | "entityId">): Application {
   207→  const audit: AuditEntry = {
   208→    ...entry,
   209→    id: genId("audit"),
   210→    entity: "Application",
   211→    entityId: app.applicationNo,
   212→    timestamp: nowISO(),
   213→  };
   214→  return { ...app, auditLog: [...app.auditLog, audit] };
   215→}
   216→
   217→function addWorkflowHistory(
   218→  app: Application,
   219→  stage: WorkflowStageKey,
   220→  actor: { name: string; role: RoleKey },
   221→  action: string,
   222→  remarks?: string,
   223→  status: WorkflowHistoryEntry["status"] = "COMPLETED"
   224→): Application {
   225→  const entry: WorkflowHistoryEntry = {
   226→    id: genId("wf"),
   227→    stage,
   228→    stageLabel: getStage(stage)?.label ?? stage,
   229→    actor,
   230→    action,
   231→    remarks,
   232→    timestamp: nowISO(),
   233→    status,
   234→  };
   235→  return { ...app, workflowHistory: [...app.workflowHistory, entry] };
   236→}
   237→
   238→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   239→  const newStage = stage ?? stageFromStatus(status);
   240→  const stageInfo = getStage(newStage);
   241→  return {
   242→    ...app,
   243→    status,
   244→    currentStage: newStage,
   245→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   246→    lastUpdated: nowISO(),
   247→    progress: computeProgress(status, newStage),
   248→  };
   249→}

   284→    const user = storeUsers.find((u) => u.role === cred.role) ?? storeUsers.find((u) => u.email === cred.email);

   286→    if (!user.active || user.status === "INACTIVE") return { ok: false, error: "Your account has been deactivated. Contact the administrator." };
   287→    if (user.status === "SUSPENDED") return { ok: false, error: "Your account has been suspended. Contact the administrator." };
   288→    if (user.status === "PENDING") return { ok: false, error: "Your account is pending approval. Please try again later." };
   289→    const portal = portalForRole(user.role);
   290→    set((s) => ({ users: s.users.map((u) => u.id === user.id ? { ...u, lastLogin: nowISO() } : u) }));

   488→        return updated;
   489→      }),
   490→    }));
   491→    // After 2.5s, complete scrutiny
   492→    setTimeout(() => {
   493→      set((s) => ({
   494→        applications: updateApp(s.applications, appId, (app) => {
   495→          const latest = app.drawings[app.drawings.length - 1];
   496→          // Deterministic: v1 fails (front setback), v2+ passes with warnings (so LTP experiences fail→reupload→pass)
   497→          const scenario = latest.version >= 2 ? "passed_warnings" : "front_setback";
   498→          const report = buildScrutinyResult(latest.version, scenario);
   499→          const passed = report.status !== "FAILED";
   500→          let updated: Application = { ...app, scrutinyReport: report };
   501→          updated = { ...updated, drawings: updated.drawings.map((d) => d.id === latest.id ? { ...d, status: passed ? "SCRUTINY_PASSED" as const : "SCRUTINY_FAILED" as const } : d) };

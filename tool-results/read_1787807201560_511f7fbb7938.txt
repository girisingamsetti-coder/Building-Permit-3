   926→   926→export function applicationsForRole(role: RoleKey): Application[] {
   927→   927→  if (role === "TPS" || role === "TPA") {
   928→   928→    return APPLICATIONS.filter((a) => ["UNDER_REVIEW", "SHORTFALL_RAISED", "DOCUMENTS_PENDING"].includes(a.status) && (a.currentStage === "TPA_TPS" || a.currentStage === "DOCUMENTS" || a.currentStage === "DRAWING_SCRUTINY"));
   929→   929→  }
   930→   930→  if (role === "ZAD" || role === "ZDD") {
   931→   931→    return APPLICATIONS.filter((a) => a.currentStage === "ZAD_ZDD");
   932→   932→  }
   933→   933→  if (role === "ZJD") {
   934→   934→    return APPLICATIONS.filter((a) => a.currentStage === "ZJD");
   935→   935→  }
   936→   936→  if (role === "DIRECTOR_DP") {
   937→   937→    return APPLICATIONS.filter((a) => a.currentStage === "DIRECTOR_DP");
   938→   938→  }
   939→   939→  if (role === "ADDL_COMMISSIONER") {
   940→   940→    return APPLICATIONS.filter((a) => a.currentStage === "ADDL_COMMISSIONER");
   941→   941→  }
   942→   942→  if (role === "COMMISSIONER") {
   943→   943→    return APPLICATIONS.filter((a) => a.currentStage === "COMMISSIONER");
   944→   944→  }
   945→   945→  return [];
   946→   946→}

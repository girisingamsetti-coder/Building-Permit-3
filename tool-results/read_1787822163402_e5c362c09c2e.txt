   155→   155→function updateApp(
   156→   156→  apps: Application[],
   157→   157→  appId: string,
   158→   158→  updater: (app: Application) => Application
   159→   159→): Application[] {
   160→   160→  return apps.map((a) => (a.id === appId ? updater(a) : a));

   163→   163→function addAudit(app: Application, entry: Omit<AuditEntry, "id" | "timestamp" | "entity" | "entityId">): Application {
   164→   164→  const audit: AuditEntry = {
   165→   165→    ...entry,
   166→   166→    id: genId("audit"),
   167→   167→    entity: "Application",
   168→   168→    entityId: app.applicationNo,
   169→   169→    timestamp: nowISO(),
   170→   170→  };
   171→   171→  return { ...app, auditLog: [...app.auditLog, audit] };
   172→   172→}

   174→   174→function addWorkflowHistory(
   175→   175→  app: Application,
   176→   176→  stage: WorkflowStageKey,
   177→   177→  actor: { name: string; role: RoleKey },
   178→   178→  action: string,
   179→   179→  remarks?: string,
   180→   180→  status: WorkflowHistoryEntry["status"] = "COMPLETED"
   181→   181→): Application {
   182→   182→  const entry: WorkflowHistoryEntry = {
   183→   183→    id: genId("wf"),
   184→   184→    stage,
   185→   185→    stageLabel: getStage(stage)?.label ?? stage,
   186→   186→    actor,
   187→   187→    action,
   188→   188→    remarks,
   189→   189→    timestamp: nowISO(),
   190→   190→    status,
   191→   191→  };
   192→   192→  return { ...app, workflowHistory: [...app.workflowHistory, entry] };
   193→   193→}

   195→   195→function setAppStatus(app: Application, status: ApplicationStatus, stage?: WorkflowStageKey): Application {
   196→   196→  const newStage = stage ?? stageFromStatus(status);
   197→   197→  const stageInfo = getStage(newStage);
   198→   198→  return {
   199→   199→    ...app,
   200→   200→    status,
   201→   201→    currentStage: newStage,
   202→   202→    currentStageLabel: stageInfo?.label ?? app.currentStageLabel,
   203→   203→    lastUpdated: nowISO(),
   204→   204→    progress: computeProgress(status, newStage),
   205→   205→  };
   206→   206→}

   838→   838→export function useVisibleApplications() {
   839→   839→  return useAppStore((s) => {
   840→   840→    if (!s.user) return [];
   841→   841→    if (s.user.role === "ADMIN") return s.applications;
   842→   842→    if (s.user.role === "LTP") return s.applications.filter((a) => a.ltpId === s.user!.id);
   843→   843→    return s.applications.filter((a) => {
   844→   844→      const roles = rolesForStage(a.currentStage);

   846→   846→      return a.workflowHistory.some((w) => w.actor.role === s.user!.role);
   847→   847→    });

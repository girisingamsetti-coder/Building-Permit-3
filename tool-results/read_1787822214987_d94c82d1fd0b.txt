   108→// ---------- Helpers ----------
   109→function daysRemaining(iso?: string): number | null {
   110→  if (!iso) return null;
   111→  const diff = new Date(iso).getTime() - Date.now();
   112→  return Math.ceil(diff / 86400000);
   113→}
   114→
   115→function slaTone(days: number | null): { cls: string; label: string } {
   116→  if (days === null) return { cls: "text-muted-foreground", label: "—" };
   117→  if (days < 0) return { cls: "text-destructive font-semibold", label: `Overdue ${Math.abs(days)}d` };
   118→  if (days <= 3) return { cls: "text-destructive font-semibold", label: `${days}d left` };
   119→  if (days <= 7) return { cls: "text-amber-600 dark:text-amber-400 font-semibold", label: `${days}d left` };
   120→  return { cls: "text-success font-medium", label: `${days}d left` };
   121→}
   122→
   123→/** Determine whether the user's role can act on the application's current stage. */
   124→function canUserActOnStage(userRole: RoleKey, stageRole: RoleKey): boolean {
   125→  if (userRole === stageRole) return true;
   126→  if (
   127→    (userRole === "TPS" || userRole === "TPA") &&
   128→    (stageRole === "TPS" || stageRole === "TPA")
   129→  )
   130→    return true;
   131→  if (
   132→    (userRole === "ZAD" || userRole === "ZDD") &&
   133→    (stageRole === "ZAD" || stageRole === "ZDD")
   134→  )
   135→    return true;
   136→  return false;
   137→}
   138→
   139→function getStageAllowedActions(app: Application): WorkflowAction[] {
   140→  const stage = WORKFLOW_STAGES.find((s) => s.key === app.currentStage);
   141→  return stage?.allowedActions ?? [];
   142→}

   943→  open,
   944→  action,
   945→  app,
   946→  onClose,
   947→}: {
   948→  open: boolean;
   949→  action: null | "APPROVE" | "FORWARD" | "RETURN" | "FINAL_DECISION";
   950→  app: Application;
   951→  onClose: () => void;

   955→  const [remarks, setRemarks] = React.useState("");
   956→  const [decision, setDecision] = React.useState<"APPROVE" | "REJECT">("APPROVE");
   957→  const [conditions, setConditions] = React.useState("");

   960→    if (open) {
   961→      setRemarks("");
   962→      setDecision("APPROVE");
   963→      setConditions("");
   964→    }
   965→  }, [open, action]);
   966→
   967→  if (!action) return null;
   968→
   969→  const stage = WORKFLOW_STAGES.find((s) => s.key === app.currentStage);
   970→  const nextStageLabel = stage?.nextStage
   971→    ? WORKFLOW_STAGES.find((s) => s.key === stage.nextStage)?.label
   972→    : undefined;
   973→
   974→  const cfg = ACTION_CONFIG[action];
   975→  const Icon = cfg.icon;
   976→
   977→  const titleMap: Record<typeof action, string> = {
   978→    APPROVE: "Approve & Forward",
   979→    FORWARD: "Forward Application",
   980→    RETURN: "Return Application",
   981→    FINAL_DECISION: "Issue Final Decision",
   982→  };
   983→
   984→  const descMap: Record<typeof action, string> = {
   985→    APPROVE: nextStageLabel
   986→      ? `Approve at the ${app.currentStageLabel} stage and forward to ${nextStageLabel}.`
   987→      : `Approve at the ${app.currentStageLabel} stage.`,

   989→      ? `Forward this application to ${nextStageLabel}.`
   990→      : `Forward this application to the next stage.`,
   991→    RETURN: "Return the application to the previous officer or applicant. A reason is required.",
   992→    FINAL_DECISION: "Issue the final approval or rejection as the Commissioner. This action is binding.",

   994→

   996→    if (action === "RETURN" && !remarks.trim()) {

  1000→      });
  1001→      return;
  1002→    }
  1003→    const actionLabel =
  1004→      action === "FINAL_DECISION"
  1005→        ? decision === "APPROVE" ? "Application approved" : "Application rejected"
  1006→        : action === "APPROVE" ? "Application approved & forwarded"
  1007→        : action === "FORWARD" ? "Application forwarded"
  1008→        : "Application returned";
  1009→
  1010→    toast({
  1011→      title: actionLabel,
  1012→      description: `${app.applicationNo} — ${action === "FINAL_DECISION" && decision === "APPROVE" ? "Final approval granted" : action === "FINAL_DECISION" ? "Final rejection issued" : "Workflow updated"}. Applicant & officers notified via SMS & in-app.`,
  1013→    });
  1014→    onClose();
  1015→    navigate("officer-applications");
  1016→  }

  1018→  return (

  1022→          <DialogTitle className="flex items-center gap-2">
  1023→            <Icon className={cn("size-5", cfg.tone)} />

  1025→          </DialogTitle>
  1026→          <DialogDescription>{descMap[action]}</DialogDescription>
  1027→        </DialogHeader>
  1028→
  1029→        <div className="space-y-4">

  1031→          <div className="rounded-lg border border-border bg-muted/30 p-3 space-y-1.5">

  1109→          {/* Remarks / reason */}

  1132→        <DialogFooter>

  1141→            className="gap-1.5"
  1142→          >

  1145→              ? decision === "APPROVE" ? "Grant Final Approval" : "Issue Final Rejection"

  1151→      </DialogContent>
  1152→    </Dialog>

  1154→}
